5. Top Goal Scorers Leaderboard (GROUP BY with HAVING) ▼
SELECT p.player_name, t.team_name, SUM(ps.points_scored) AS total_goals, COUNT(ps.match_id) AS matches_played FROM PlayerMatchStat ps JOIN Player p ON ps.player_id = p.player_id JOIN Team t ON p.team_id = t.team_id WHERE ps.played = 1 GROUP BY p.player_id, p.player_name, t.team_name HAVING total_goals > 0 ORDER BY total_goals DESC;

10. Department-wise Total Goals Scored (Nested Multi-Table Aggregation) ▼
SELECT d.dept_name, t.team_name, COALESCE(SUM(ps.points_scored), 0) AS total_goals FROM Department d JOIN Team t ON d.dept_id = t.dept_id LEFT JOIN Player p ON t.team_id = p.team_id LEFT JOIN PlayerMatchStat ps ON p.player_id = ps.player_id AND ps.played = 1 GROUP BY d.dept_name, t.team_name ORDER BY total_goals DESC;
